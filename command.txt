# ssh_public_key = "<contents of sftp-key.pub>"
# 3. Connect:
# sftp -i sftp-key <SFTP_USERNAME>@<SFTP_ENDPOINT>

# =============================================================================

chmod 600 sftp-key.pem
sftp -i sftp-key.pem <SFTP_USERNAME>@<SFTP_ENDPOINT>
